# Ontrak Sync backend.
#
# Deliberately short. The Network's other Python services bring a framework and a
# dozen libraries; this one needs a web framework, an ASGI server and nothing else,
# because everything it does to the Network runs over SSH and the os/apt output
# parsers are hand-written (and unit-tested) rather than imported. Fewer
# dependencies is less to patch — which matters for a tool whose job is patching.
#
# No pytest: the tests are plain `unittest` and run with the interpreter, so the
# container that runs the service needs no test tooling at all.

fastapi==0.121.2
uvicorn[standard]==0.42.0

# Cerulean SSO. Verifying an RS256 ID token means RSA signature verification, and
# there is no way to do that correctly with the standard library — hand-rolling
# bignum modular exponentiation to check a signature is exactly the kind of thing
# that looks fine and accepts a forgery. Everything else about the OIDC flow
# (discovery, PKCE, the token exchange, claim checks) is hand-written in
# `ontrak/oidc.py` and unit-tested, so this is the whole of the SSO dependency
# surface: one well-audited library used for one operation.
cryptography>=43.0.0
